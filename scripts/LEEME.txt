-- Registro inicial de bitácora para probar el formulario

DECLARE @IDAdmin INT;

SELECT @IDAdmin = IDUser
FROM dbo.Usuario
WHERE NombreUsuario = 'admin';

IF @IDAdmin IS NOT NULL
AND NOT EXISTS (
    SELECT 1
    FROM dbo.Bitacora
    WHERE IDUser = @IDAdmin
      AND Actividad = 'Login exitoso'
)
BEGIN
    INSERT INTO dbo.Bitacora (IDUser, Actividad, Fecha, Hora)
    VALUES (
        @IDAdmin,
        'Login exitoso',
        CAST(GETDATE() AS DATE),
        CAST(GETDATE() AS TIME)
    );
END
GO
